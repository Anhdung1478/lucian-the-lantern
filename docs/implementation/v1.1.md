# V1.1: Skin effects for Q, W, E and R — Implementation Plan

> **Project:** `E:\Project\lucian-the-latern` (Godot 4.7.2 game).
> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Each themed skin (High Noon, PROJECT, Pulsefire, Vampire) gets its own look and sounds for Q, W, E, R, basic attacks and the muzzle flash. Classic keeps 1.0's effects. The Skin Shop gets a Preview pop-up that plays a skin's effects.

**Architecture:**
- A `SkinFx` resource per skin builds that skin's effect nodes and names its sounds. `SkinData.fx` points at it.
- The skills and `Lucian` ask `lucian.fx()` for every effect node and sound instead of making `LaserBeam`, `CrossBlast` and the rest themselves. Damage, hit shapes and timing stay in the skills.
- Projectiles keep their hit logic and show a swappable `ShotLook` child.
- Shared bases (`FxPiece`, `FxBeam`, `FxBlast`, `FxTrail`, `FxSparks`) hold timing, Low effects and particles.

**Tech stack:** Godot 4.7.2, GDScript, GUT tests, all art drawn in code with `_draw()`, sounds synthesized in `Sfx`.

**Spec:** `docs/superpowers/specs/2026-10-30-skin-effects-design.md`. Read it with this plan: it has the look tables (Section 1) and the sound tables (Section 3).

## Global constraints

- **Code style:** Godot 4.7.2 GDScript in the style of the surrounding code:
  - tabs, typed vars, `##` doc comments, `&"name"` StringNames;
  - one `class_name` per file;
  - no autoloads and no new `.tscn`.
- **No gameplay change:** never change damage, hit shapes, cast times, cooldowns, ranges or speeds.
- **Main shapes match the hitboxes:** Q's beam is `length × width` and W's cross arms are `bar`.
- **Main shapes last as long as Classic's:**

- **Low effects:** only the main shapes; no particles, rings, bats, afterimages, scan lines or pixel break-up. No muzzle flash or dash trail is made, as in 1.0.

- **Others keep their looks:** Senna's shots, item effects (Galeforce, Rocketbelt, Runaan's), Dark Lucian's Q and trail, and the Act 1 E-lesson trail.

```gdscript
class_name FxSparks
extends RefCounted
## Small particles for skin effects (embers, drops, pixels, sparks), in the
## frame of the node that draws them. Each lives at most MAX_LIFE seconds,
## so no effect outlasts 0.6 s.

enum Shape { DOT, SQUARE, HEX, STREAK }

const MAX_LIFE := 0.6

## Added to every velocity each second (falling drops).
var gravity := Vector2.ZERO
## Share of the velocity lost each second (embers slowing down).
var drag := 0.0

## Each: {at, velocity, life, left, size, color}.
var _items: Array[Dictionary] = []


func emit(at: Vector2, velocity: Vector2, life: float, size: float, color: Color) -> void:
	var capped := minf(life, MAX_LIFE)
	_items.append({"at": at, "velocity": velocity, "life": capped, "left": capped, "size": size, "color": color})


func step(delta: float) -> void:
	for item in _items:
		item.left -= delta
		item.velocity += gravity * delta
		item.velocity *= maxf(1.0 - drag * delta, 0.0)
		item.at += item.velocity * delta
	_items.assign(_items.filter(func(item: Dictionary) -> bool: return item.left > 0.0))


func size() -> int:
	return _items.size()


func is_empty() -> bool:
	return _items.is_empty()


## Draws them on `canvas`, from inside its _draw, each fading as it ages.
func draw_on(canvas: CanvasItem, shape := Shape.DOT) -> void:
	for item in _items:
		var color: Color = item.color
		color.a *= item.left / item.life
		var at: Vector2 = item.at
		var side: float = item.size
		match shape:
			Shape.SQUARE:
				canvas.draw_rect(Rect2(at - Vector2.ONE * side * 0.5, Vector2.ONE * side), color)
			Shape.HEX:
				var points := PackedVector2Array()
				for i in 6:
					points.append(at + Vector2.RIGHT.rotated(i * PI / 3.0) * side)
				canvas.draw_colored_polygon(points, color)
			Shape.STREAK:
				var velocity: Vector2 = item.velocity
				canvas.draw_line(at, at - velocity * 0.03, color, side)
			_:
				canvas.draw_circle(at, side, color)
```

```gdscript
class_name FxPiece
extends Node2D
## A skin effect that shows its main shape for `lifetime` seconds and frees
## itself once its details are gone, never later than FxSparks.MAX_LIFE.
## With Low effects (`low`, read when it enters the tree) it makes no
## particles or other details: only the main shape.

var lifetime := 0.25
var age := 0.0
var low := false
var sparks := FxSparks.new()


func _ready() -> void:
	low = GameSettings.instance().low_effects()
	if not low:
		_start()


## Makes the first particles and details. Never called on Low effects.
func _start() -> void:
	pass


func _process(delta: float) -> void:
	age += delta
	sparks.step(delta)
	if age >= FxSparks.MAX_LIFE or (age >= lifetime and sparks.is_empty() and not _lingers()):
		queue_free()
		return
	queue_redraw()


## True while a detail other than particles (a scorch ring, bats) still shows.
func _lingers() -> bool:
	return false


## How many particles and other details (rings, bats, scan lines) show now.
## Always 0 on Low effects.
func details() -> int:
	return sparks.size()


## 0 → 1 over `lifetime`.
func progress() -> float:
	return clampf(age / lifetime, 0.0, 1.0)


## The main shape's alpha: 1 → 0 over `lifetime`, like the linear fade of
## LaserBeam and CrossBlast.
func fade() -> float:
	return 1.0 - progress()
```

```gdscript
class_name ShotLook
extends Node2D
## How one of Lucian's shots looks, drawn facing +x in its own frame: W's
## missile, an R bullet or a basic attack bolt. A projectile given a look
## (set_look) adds it as a child and stops drawing itself. The static
## helpers are the projectiles' own plain drawing, shared with PlainShotLook.

enum Kind { MISSILE, BULLET, BOLT }

var kind := Kind.BULLET
## Seconds since it was added, for looks that animate.
var age := 0.0
## Low effects, read when it enters the tree: no flicker or spin.
var low := false


func _init(shot_kind := Kind.BULLET) -> void:
	kind = shot_kind


func _ready() -> void:
	low = GameSettings.instance().low_effects()


## A streak with a glow, and a round head when `head_radius` > 0 (W's
## fireball): LinearProjectile's own look.
static func draw_streak(canvas: CanvasItem, color: Color, glow_color: Color, streak_length: float, streak_width: float,
		head_radius: float) -> void:
	canvas.draw_line(Vector2(-streak_length * 1.4, 0), Vector2.ZERO, glow_color, streak_width * 2.6)
	canvas.draw_line(Vector2(-streak_length, 0), Vector2(streak_width * 0.5, 0), color, streak_width)
	if head_radius > 0.0:
		canvas.draw_circle(Vector2.ZERO, head_radius * 1.6, glow_color)
		canvas.draw_circle(Vector2.ZERO, head_radius, color)
		canvas.draw_circle(Vector2.ZERO, head_radius * 0.5, Color.WHITE)

class_name ProjectBeam
extends FxBeam
## PROJECT's Q: a hologram beam. Bright cyan edge lines over a see-through
## fill and scan lines sweeping along it. In its second half it breaks into
## pixels that wink out one by one, instead of fading. On Low effects it
## fades whole.

const FILL := Color(0.22, 0.91, 1.0, 0.2)
const EDGE := Color("7ff4ff")
const CENTER := Color("f0ffff")
const SCAN := Color(0.5, 0.96, 1.0, 0.5)
const PIXEL := Color(0.22, 0.91, 1.0, 0.55)
const SCAN_LINES := 5
## Scan lines sweep this many px a second.
const SCAN_SPEED := 2400.0
## Pixel rows across the beam.
const ROWS := 3
const SPARKS := 16

## Each pixel's share of the lifetime when it winks out (0.5–1).
var _pixel_ends := PackedFloat32Array()
var _columns := 0


func _start() -> void:
	_columns = maxi(ceili(length / (width / ROWS)), 1)
	for i in _columns * ROWS:
		_pixel_ends.append(randf_range(0.5, 1.0))
	for i in SPARKS:
		var side := 1.0 if randf() < 0.5 else -1.0
		sparks.emit(Vector2(randf_range(0.0, length), side * width * 0.45), Vector2(0.0, side * randf_range(20.0, 80.0)),
			randf_range(0.3, 0.45), randf_range(3.0, 5.0), EDGE)


func details() -> int:
	return sparks.size() + (SCAN_LINES if not low and age < lifetime else 0)


func _draw() -> void:
	if age < lifetime:
		if low:
			_draw_whole(fade())
		elif progress() < 0.5:
			_draw_whole(1.0)
		else:
			_draw_pixels()
		if not low:
			_draw_scan_lines()
	sparks.draw_on(self, FxSparks.Shape.SQUARE)


func _draw_whole(a: float) -> void:
	draw_rect(Rect2(0, -width * 0.5, length, width), Color(FILL, FILL.a * a))
	draw_line(Vector2(0, -width * 0.5), Vector2(length, -width * 0.5), Color(EDGE, a), 3.0)
	draw_line(Vector2(0, width * 0.5), Vector2(length, width * 0.5), Color(EDGE, a), 3.0)
	draw_rect(Rect2(0, -width * 0.06, length, width * 0.12), Color(CENTER, a))


func _draw_pixels() -> void:
	var cell := width / ROWS
	var p := progress()
	for column in _columns:
		var x := column * cell
		var side := minf(cell, length - x) - 1.0
		for row in ROWS:
			if p < _pixel_ends[column * ROWS + row]:
				draw_rect(Rect2(x, -width * 0.5 + row * cell, side, cell - 1.0), PIXEL)


func _draw_scan_lines() -> void:
	for i in SCAN_LINES:
		var x := fmod(age * SCAN_SPEED + i * length / SCAN_LINES, length)
		draw_line(Vector2(x, -width * 0.5), Vector2(x, width * 0.5), SCAN, 2.0)
```

class_name ProjectBlast
extends FxBlast
## PROJECT's W: a cross built from grid squares that light up from the
## centre outward, then flicker off one by one, with pixel sparks flying
## from the arm ends. On Low effects: a plain cyan cross that fades.

const CELL_FILL := Color(0.22, 0.91, 1.0, 0.55)
const CELL_EDGE := Color("7ff4ff")
const CORE := Color("f0ffff")
const PLAIN_OUTER := Color(0.22, 0.91, 1.0, 0.45)
const PLAIN_INNER := Color(0.5, 0.96, 1.0, 0.8)
## Seconds the light-up takes to reach the arm ends.
const SPREAD_TIME := 0.12
const SPARKS_PER_END := 3

## Each cell: [centre, side, lit at, off at] (seconds).
var _cells: Array[Array] = []


func _start() -> void:
	var side := bar.y / 3.0
	var reach := bar.x * 0.5
	for arm: Vector2 in [bar, Vector2(bar.y, bar.x)]:
		for column in roundi(arm.x / side):
			for row in roundi(arm.y / side):
				var center := -arm * 0.5 + Vector2(column + 0.5, row + 0.5) * side
				if _has_cell(center):
					continue
				var lit_at := center.length() / reach * SPREAD_TIME
				_cells.append([center, side, lit_at, randf_range(maxf(lit_at + 0.08, 0.2), lifetime)])
	for direction: Vector2 in [Vector2.RIGHT, Vector2.DOWN, Vector2.LEFT, Vector2.UP]:
		for i in SPARKS_PER_END:
			sparks.emit(direction * reach, direction.rotated(randf_range(-0.6, 0.6)) * randf_range(120.0, 260.0),
				randf_range(0.25, 0.4), 4.0, CELL_EDGE)


func details() -> int:
	return sparks.size() + _cells.filter(func(cell: Array) -> bool: return age >= cell[2] and age < cell[3]).size()


func _has_cell(center: Vector2) -> bool:
	for cell in _cells:
		if (cell[0] as Vector2).is_equal_approx(center):
			return true
	return false


func _draw() -> void:
	if low:
		var a := fade()
		if a > 0.0:
			for arm in [bar, Vector2(bar.y, bar.x)]:
				draw_rect(Rect2(-arm * 0.5, arm), Color(PLAIN_OUTER, PLAIN_OUTER.a * a))
				draw_rect(Rect2(-arm * 0.25, arm * 0.5), Color(PLAIN_INNER, PLAIN_INNER.a * a))
		return
	for cell in _cells:
		if age >= cell[2] and age < cell[3]:
			var side: float = cell[1]
			var rect := Rect2(cell[0] - Vector2.ONE * (side * 0.5 - 1.0), Vector2.ONE * (side - 2.0))
			draw_rect(rect, CELL_FILL)
			draw_rect(rect, CELL_EDGE, false, 1.0)
	if age < lifetime:
		var d := bar.y * 0.3 * fade()
		draw_colored_polygon(PackedVector2Array([Vector2(d, 0), Vector2(0, d), Vector2(-d, 0), Vector2(0, -d)]), CORE)
	sparks.draw_on(self, FxSparks.Shape.SQUARE)

## into a cyan and a magenta image that jitter, over broken scan-line pieces.

## Seconds between glitches (new scan-line pieces and jitter).
const GLITCH_EVERY := 0.05
## The cyan and magenta images sit this far apart, each side.

const JITTER := 4.0

var _next_glitch := 0.0


func _process(delta: float) -> void:
	super(delta)
	if age >= _next_glitch:
		_next_glitch = age + GLITCH_EVERY
		_glitch()


func _glitch() -> void:
	_pieces.clear()
	for i in PIECES:
		var start := randf()
		_pieces.append(Vector3(randf_range(-20.0, 20.0), start, minf(start + randf_range(0.08, 0.25), 1.0)))
	for i in _jitter.size():
		_jitter[i] = randf_range(-JITTER, JITTER)


func _draw() -> void:
	draw_rect(Rect2(0, -20, length, 40), BAND)
	for piece in _pieces:
		draw_line(Vector2(piece.y * length, piece.x), Vector2(piece.z * length, piece.x), LINE, 2.0)
	if body == null:
		return
	var half := body.get_size() * 0.5
	for i in COPIES.size():
		var at := Vector2(COPIES[i] * length + _jitter[i], 0)
		draw_texture(body, at - half + Vector2(0, -SPLIT), CYAN)
		draw_texture(body, at - half + Vector2(0, SPLIT), MAGENTA)

class_name PulsefireRing
extends RefCounted
## Drawing helpers for Pulsefire's time-rift look: tilted rings (ellipses)
## and a clock face with a turning hand.


## An ellipse of `radii` around `center`, turned by `angle`, closed.
static func ellipse(center: Vector2, radii: Vector2, angle: float, segments := 24) -> PackedVector2Array:
	var points := PackedVector2Array()
	for i in segments + 1:
		var t := TAU * i / segments
		points.append(center + Vector2(cos(t) * radii.x, sin(t) * radii.y).rotated(angle))
	return points


## A clock face on `canvas`: a ring, 12 ticks and one hand at `hand` radians.
static func draw_clock(canvas: CanvasItem, center: Vector2, radius: float, hand: float, color: Color) -> void:
	canvas.draw_arc(center, radius, 0.0, TAU, 32, color, 2.0)
	for i in 12:
		var direction := Vector2.RIGHT.rotated(TAU * i / 12.0)
		canvas.draw_line(center + direction * radius * 0.78, center + direction * radius, color, 1.5)
	canvas.draw_line(center, center + Vector2.RIGHT.rotated(hand) * radius * 0.7, color, 2.0)

class_name VampireBat
extends RefCounted
## A small bat for Vampire's W cross and E trail, seen from above: its body
## along its heading, its wings out to the sides.

const BODY := Color(0.07, 0.02, 0.04, 0.95)
const RIM := Color("ff3a50")


## Draws a bat on `canvas` at `at`, heading `heading` radians, `span` wide.
## `flap` (-1..1) opens and closes the wings; `alpha` fades it.
static func draw(canvas: CanvasItem, at: Vector2, heading: float, span: float, flap: float, alpha := 1.0) -> void:
	var tip := span * 0.5 * (0.65 + 0.35 * flap)
	var shape: Array[Vector2] = [
		Vector2(0.22 * span, 0.0), Vector2(0.05 * span, -0.07 * span), Vector2(0.0, -tip),
		Vector2(-0.08 * span, -tip * 0.6), Vector2(-0.14 * span, -tip * 0.85), Vector2(-0.18 * span, -0.06 * span),
		Vector2(-0.26 * span, 0.0), Vector2(-0.18 * span, 0.06 * span), Vector2(-0.14 * span, tip * 0.85),
		Vector2(-0.08 * span, tip * 0.6), Vector2(0.0, tip), Vector2(0.05 * span, 0.07 * span),
	]
	var points := PackedVector2Array()
	for point in shape:
		points.append(at + point.rotated(heading))
	canvas.draw_colored_polygon(points, Color(BODY, BODY.a * alpha))
	points.append(points[0])
	canvas.draw_polyline(points, Color(RIM, alpha), 1.5)

```gdscript
class_name SkinPreview
extends Control
## The Skin Shop's preview of one skin's effects: Lucian in that skin and a
## training dummy on a stage floor. Q, W, E, R and basic attacks play with
## the skin's SkinFx at game size and with its sounds. ▶ All loops them and
## starts on opening; a skill's button plays only that one. It deals no
## damage and saves nothing. ✕ or ESC (SkinShop.close_popup) closes it and
## frees everything it spawned.

signal closed

const PANEL_SIZE := Vector2(1200, 680)
const STAGE_SIZE := Vector2(1140, 500)
const LUCIAN_AT := Vector2(220, 250)
const DUMMY_AT := Vector2(920, 250)
## The gun muzzles of lucian.tscn, facing right: his own gun, then Senna's.
const MUZZLES: Array[Vector2] = [Vector2(46, -15), Vector2(46, 15)]
## Rank 1 numbers, as in the skills (a test keeps them equal).
const Q_CAST := 0.3
const Q_WIDTH := 60.0
const W_CAST := 0.2
const W_SPEED := 1600.0
const W_BAR := Vector2(400, 60)
const E_DISTANCE := 280.0
const E_SPEED := 1350.0
const R_SHOTS := 22
const R_DURATION := 3.0
const R_SPEED := 2800.0
const ATTACK_SHOTS := 3
const ATTACK_GAP := 0.4
## Seconds after a piece before the next one in ▶ All.
const GAP := 0.6
## Seconds Lucian takes to walk back after E.
const WALK_BACK := 0.8
## ▶ All's order; each piece has a button.
const PIECES: Array[StringName] = [&"q", &"w", &"e", &"r", &"attack"]
const LABELS := {&"q": "Q", &"w": "W", &"e": "E", &"r": "R", &"attack": "Attack", &"all": "▶ All"}
const DUMMY_ART := preload("res://assets/art/enemies/training_dummy.svg")
const FLOOR := Color(0.08, 0.09, 0.12)
const TILE_LINE := Color(0.13, 0.15, 0.19)
const TILE := 80.0

var skin: SkinData
## Piece id (and &"all") → its button.
var buttons := {}
var close_button := Button.new()
## The floor, clipped to STAGE_SIZE. Lucian, the dummy and the effects stand on it.
var stage := Control.new()
var lucian := Sprite2D.new()
var dummy := Sprite2D.new()
var effects := Node2D.new()

var _fx: SkinFx
var _tween: Tween
var _gun := 0
var _trail: DashTrail


func _init(skin_data: SkinData = null) -> void:
	skin = skin_data
	_fx = skin.fx if skin and skin.fx else SkinFx.new()


func _ready() -> void:
	set_anchors_and_offsets_preset(Control.PRESET_FULL_RECT)
	_build()
	play_all()


## Plays one piece (&"q", &"w", &"e", &"r" or &"attack"), stopping ▶ All.
func play(piece: StringName) -> void:
	_stop()
	_tween = create_tween()
	_queue(piece)


## Plays every piece in turn, with a gap, over and over.
func play_all() -> void:
	_stop()
	_tween = create_tween().set_loops()
	for piece in PIECES:
		_queue(piece)
		_tween.tween_interval(GAP)


## Closes the preview and frees everything it spawned.
func close() -> void:
	_stop()
	closed.emit()
	queue_free()


func _process(_delta: float) -> void:
	if is_instance_valid(_trail):
		_trail.reach(lucian.position)


func _queue(piece: StringName) -> void:
	match piece:
		&"q":
			_tween.tween_interval(Q_CAST)
			_tween.tween_callback(_fire_q)
			_tween.tween_interval(LaserBeam.LIFETIME)
		&"w":
			_tween.tween_interval(W_CAST)
			_tween.tween_callback(_fire_w)
			_tween.tween_interval((DUMMY_AT.x - LUCIAN_AT.x - Lucian.GUN_REACH) / W_SPEED + CrossBlast.LIFETIME)
		&"e":
			var landing := LUCIAN_AT + Vector2(E_DISTANCE, 0)
			_tween.tween_callback(_start_dash)
			_tween.tween_property(lucian, "position", landing, E_DISTANCE / E_SPEED).from(LUCIAN_AT)
			_tween.tween_callback(_end_dash)
			_tween.tween_interval(DashTrail.LIFETIME)
			_tween.tween_property(lucian, "position", LUCIAN_AT, WALK_BACK).from(landing)
		&"r":
			for i in R_SHOTS:
				_tween.tween_callback(_shoot.bind(_fx.r_bullet_look, R_SPEED))
				_tween.tween_interval(R_DURATION / R_SHOTS)
		&"attack":
			for i in ATTACK_SHOTS:
				_tween.tween_callback(_shoot.bind(_fx.attack_shot_look, AttackProjectile.SPEED))
				_tween.tween_interval(ATTACK_GAP)


func _fire_q() -> void:
	var from := LUCIAN_AT + Vector2(Lucian.GUN_REACH, 0)
	effects.add_child(_fx.q_beam(from, Vector2.RIGHT, DUMMY_AT.x - from.x, Q_WIDTH))
	Sfx.play(_fx.q_sound)


func _fire_w() -> void:
	var from := LUCIAN_AT + Vector2(Lucian.GUN_REACH, 0)
	var missile := LinearProjectile.new()
	missile.launch(from, Vector2.RIGHT, W_SPEED, DUMMY_AT.x - from.x, 16.0)
	missile.set_look(_fx.w_missile_look())
	missile.finished.connect(_burst, CONNECT_DEFERRED)
	effects.add_child(missile)
	Sfx.play(_fx.w_sound)


## W's cross where the missile ended (`at` is a global position).
func _burst(at: Vector2) -> void:
	effects.add_child(_fx.w_blast(effects.to_local(at), Vector2.RIGHT, W_BAR))


func _start_dash() -> void:
	Sfx.play(_fx.e_sound)
	if GameSettings.instance().low_effects():
		return
	_trail = _fx.e_trail(skin.body)
	_trail.setup(LUCIAN_AT, DashTrail.LIGHT)
	effects.add_child(_trail)


func _end_dash() -> void:
	if is_instance_valid(_trail):
		_trail.reach(lucian.position)
		_trail.finish()
	_trail = null


## A shot from the next gun to the dummy, with its muzzle flash and sound.
func _shoot(make_look: Callable, speed: float) -> void:
	var from := LUCIAN_AT + MUZZLES[_gun]
	_gun = 1 - _gun
	var shot := LinearProjectile.new()
	shot.launch(from, Vector2.RIGHT, speed, DUMMY_AT.x - from.x, 10.0)
	shot.set_look(make_look.call())
	effects.add_child(shot)
	Sfx.play(_fx.shot_sound, -6.0)
	if not GameSettings.instance().low_effects():
		var flash := _fx.muzzle_flash()
		flash.position = from
		effects.add_child(flash)


## Stops what is playing and puts Lucian back. Effects already out finish
## on their own.
func _stop() -> void:
	if _tween:
		_tween.kill()
	_tween = null
	_end_dash()
	lucian.position = LUCIAN_AT


func _build() -> void:
	# Clicks outside the panel stop here.
	mouse_filter = Control.MOUSE_FILTER_STOP
	var backdrop := ColorRect.new()
	backdrop.color = Color(0, 0, 0, 0.55)
	backdrop.set_anchors_preset(Control.PRESET_FULL_RECT)
	backdrop.mouse_filter = Control.MOUSE_FILTER_IGNORE
	add_child(backdrop)
	var center := CenterContainer.new()
	center.set_anchors_preset(Control.PRESET_FULL_RECT)
	center.mouse_filter = Control.MOUSE_FILTER_IGNORE
	add_child(center)
	var panel := PanelContainer.new()
	panel.custom_minimum_size = PANEL_SIZE
	panel.add_theme_stylebox_override(&"panel", UiStyle.panel(UiStyle.PANEL_COLOR, UiStyle.BORDER_COLOR, 2, 24.0))
	center.add_child(panel)
	var column := VBoxContainer.new()
	column.add_theme_constant_override(&"separation", 14)
	panel.add_child(column)

	var header := HBoxContainer.new()
	column.add_child(header)
	var title := UiStyle.label("%s · Preview" % skin.display_name, 26, UiStyle.TITLE_COLOR)
	title.size_flags_horizontal = Control.SIZE_EXPAND_FILL
	header.add_child(title)
	close_button.text = "✕"
	UiStyle.style_button(close_button, 22)
	close_button.custom_minimum_size = Vector2(48, 48)
	close_button.pressed.connect(close)
	header.add_child(close_button)

	stage.custom_minimum_size = STAGE_SIZE
	stage.clip_contents = true
	stage.draw.connect(_draw_floor)
	column.add_child(stage)
	dummy.texture = DUMMY_ART
	dummy.position = DUMMY_AT
	stage.add_child(dummy)
	lucian.texture = skin.body
	lucian.position = LUCIAN_AT
	stage.add_child(lucian)
	stage.add_child(effects)

	var row := HBoxContainer.new()
	row.alignment = BoxContainer.ALIGNMENT_CENTER
	row.add_theme_constant_override(&"separation", 14)
	column.add_child(row)
	for piece: StringName in PIECES + [&"all"]:
		var button := Button.new()
		button.text = LABELS[piece]
		UiStyle.style_button(button, 20)
		button.custom_minimum_size = Vector2(130, 48)
		button.pressed.connect(play_all if piece == &"all" else play.bind(piece))
		row.add_child(button)
		buttons[piece] = button


func _draw_floor() -> void:
	stage.draw_rect(Rect2(Vector2.ZERO, STAGE_SIZE), FLOOR)
	var x := TILE
	while x < STAGE_SIZE.x:
		stage.draw_line(Vector2(x, 0), Vector2(x, STAGE_SIZE.y), TILE_LINE, 1.0)
		x += TILE
	var y := TILE
	while y < STAGE_SIZE.y:
		stage.draw_line(Vector2(0, y), Vector2(STAGE_SIZE.x, y), TILE_LINE, 1.0)
		y += TILE
```
