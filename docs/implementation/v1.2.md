# V1.2: Evolution Tree, node states, skin purchase confirmation and menus — Implementation Plan

> **Project:** `E:\Project\lucian-the-latern` (Godot 4.7.2 game).
> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** The Evolution Tree becomes a living lantern tree, its nodes show "not bought", "in progress" and "done" at a glance, buying a skin asks first, and the stage and main-screen menus become ornate cards.

**Architecture:**
- **Tree drawing** moves out of `TreePage` into a new `TreeArt`. `TreeArt` lays out the trunk, roots, boughs, canopy, lanterns and light paths once per page, then draws them.
- **Node states:** `TreeNodeView` keeps its state ids and only changes the look: greyscale or colour, silver / gold / green, a level bar, chips.
- **Shared menu pieces:** `MenuBackdrop`, `MenuCard` and `MenuEntry` build `GameMenu`, `GearMenu`, `ConfirmPopup` and the new `SkinPurchasePopup`.

**Tech stack:** Godot 4.7.2, GDScript, GUT tests. All art is drawn in code with `_draw()` or is hand-written SVG.

**Spec:** `docs/superpowers/specs/2026-11-04-v1.2-tree-and-menus-design.md`. Read it with this plan.

class_name TreeNodeView
extends Control
## One Evolution Tree node on its branch page: a 96 px square frame with the
## node's icon, a level badge tucked under it and the node's name below. The
## look tells its progress at a glance:
## - hidden: a "?" on a dark frame;
## - not bought (open, or poor when too dear): a greyscale icon in a thin
##   silver frame, its level bar empty;
## - in progress (bought): full colour in a gold frame with a gold glow, the
##   bar lit gold up to the level;
## - done (maxed): a green frame and glow, the bar all green, "MAX" and a
##   check crest at the top-right corner.
## A gold ▲ at the top-left says the next level can be bought now (FREE
## shows at the top-right instead when it costs nothing); a lock at the
## bottom-right says a bought node's next level is closed. A left press
## picks it.

signal chosen(id: StringName)

const SIZE := 96.0
const ICON_INSET := 6.0
const CORNER := 10
## The frame's border: thick once bought, thin before.
const BORDER := 3
const THIN_BORDER := 2
## The level badge: a pill tucked under the frame, its number on top and a
## level bar along its bottom. How far it tucks under, its font and the
## number's height; the bar's height, its gap to the pill's bottom, its side
## inset and the gap between segments.
const BADGE_SIZE := Vector2(76, 32)
const BADGE_TUCK := 14.0
const BADGE_FONT := 17
const BADGE_TEXT_HEIGHT := 23.0
const BAR_HEIGHT := 4.0
const BAR_BOTTOM := 5.0
const BAR_INSET := 10.0
const BAR_GAP := 2.0
## The name under the badge, on a dark plate so boughs and vines never cross
## it: its font, the gap above it and the plate's padding.
const NAME_FONT := 17
const NAME_GAP := 4.0
const NAME_PAD := Vector2(8, 1)
const NAME_PLATE := Color(0.015, 0.025, 0.03, 0.78)
## The lock at the bottom-right corner.
const LOCK_RADIUS := 13.0
const LOCK_AT := Vector2(SIZE - 4.0, SIZE - 12.0)
## The ready ▲ (top-left) and the crest (top-right): their radius and where
## their middles sit, poking past the corners like FREE.
const CHIP_RADIUS := 12.0
const READY_AT := Vector2(2.0, 2.0)
const CREST_AT := Vector2(SIZE - 2.0, 2.0)
## The FREE badge at the top-right corner: how far it pokes past the frame.
const FREE_POKE := Vector2(10.0, 10.0)
const LOCK := preload("res://assets/art/ui/lock.svg")
const GREY_SHADER := preload("res://src/ui/hud/grey_icon.gdshader")
const HIDDEN_COLOR := Color("2a2f3a")
const SILVER_COLOR := Color("aeb6c4")
const BOUGHT_COLOR := UiStyle.GOLD_COLOR
const MAXED_COLOR := Color("6fd36f")
const QUESTION_COLOR := Color("6b7280")
const FILL_COLOR := Color("10141b")
const HIDDEN_FILL := Color("0b0e13")
const BADGE_FILL := Color(0.03, 0.04, 0.06, 0.94)
const SEGMENT_OFF := Color(1, 1, 1, 0.13)
const CHIP_FILL := Color(0.04, 0.05, 0.07, 0.95)

## Shared by every greyscale icon (grey_material()).
static var _grey: ShaderMaterial

var data: TreeNodeData
var tree: EvolutionTree
var essences: Essences
## Drawn with an extra white outline.
var selected := false:
	set(value):
		selected = value
		queue_redraw()
## The level's number ("2/5", "MAX"); not shown on a hidden node.
var badge: Label
## At the top-right corner while the next level is free.
var free_badge := FreeBadge.new()
## The gold ▲ at the top-left: the next level can be bought now.
var ready_chip := Control.new()
## The green check at the top-right: every level bought.
var crest := Control.new()

var _icon: TextureRect
var _question: Label
## The pill under the frame and its level bar; the badge's number is over it.
var _pill := Control.new()
## Over the icon, so it is a child of its own.
var _lock := Control.new()
var _hovered := false


func _init(node_data: TreeNodeData, evolution_tree: EvolutionTree, owned: Essences) -> void:
	data = node_data
	tree = evolution_tree
	essences = owned
	size = Vector2(SIZE, SIZE + BADGE_SIZE.y - BADGE_TUCK)

	badge.vertical_alignment = VERTICAL_ALIGNMENT_CENTER

	crest.draw.connect(_draw_crest)

## Whether node `id` shows as a "?": not open and nothing bought yet (a
## level bought through the debug tools still shows).
static func is_hidden(evolution_tree: EvolutionTree, id: StringName) -> bool:
	return evolution_tree.level(id) == 0 and not evolution_tree.is_open(id)


## &"hidden", &"open", &"poor" (open but too dear), &"bought" (more levels
## left) or &"maxed", for node `id`.
static func state_of(evolution_tree: EvolutionTree, owned: Essences, id: StringName) -> StringName:
	if is_hidden(evolution_tree, id):
		return &"hidden"
	if evolution_tree.is_maxed(id):
		return &"maxed"
	if evolution_tree.level(id) > 0:
		return &"bought"
	return &"open" if evolution_tree.can_buy(id, owned) else &"poor"


## The frame's border for a state: silver until bought, gold in progress,
## green when done.
static func state_color(state_name: StringName) -> Color:
	match state_name:
		&"hidden":
			return HIDDEN_COLOR
		&"bought":
			return BOUGHT_COLOR
		&"maxed":
			return MAXED_COLOR
	return SILVER_COLOR


## The frame's border width for a state: thick once bought.
static func border_width(state_name: StringName) -> int:
	return BORDER if state_name in [&"bought", &"maxed"] else THIN_BORDER


## The greyscale look of an icon not bought yet, shared.
static func grey_material() -> ShaderMaterial:
	if _grey == null:
		_grey = ShaderMaterial.new()
		_grey.shader = GREY_SHADER
	return _grey


## The level bar's segments for `count` levels, in the pill's coordinates.
static func segment_rects(count: int) -> Array[Rect2]:
	var out: Array[Rect2] = []
	var width := (BADGE_SIZE.x - BAR_INSET * 2.0 - BAR_GAP * (count - 1)) / count
	var y := BADGE_SIZE.y - BAR_BOTTOM - BAR_HEIGHT
	for i in count:
		out.append(Rect2(BAR_INSET + i * (width + BAR_GAP), y, width, BAR_HEIGHT))
	return out


func state() -> StringName:
	return state_of(tree, essences, data.id)


## Bought, but its next level is closed: the lock shows at the corner.
func locked() -> bool:
	return tree.level(data.id) > 0 and not tree.is_maxed(data.id) and not tree.next_level_open(data.id)


## The next level can be bought now and costs something: the ▲ shows.
func can_upgrade() -> bool:
	return tree.can_buy(data.id, essences) and not tree.is_free(data.id)


## The icon shows in greyscale: open or too dear, nothing bought yet.
func is_grey() -> bool:
	return _icon.material != null


## The name under the badge: "???" while hidden.
func shown_name() -> String:
	return "???" if state() == &"hidden" else data.display_name


## Everything this view draws (frame, badge and name), in its own coordinates;
## the name may be wider than the frame.
func footprint() -> Rect2:
	var plate := _name_plate()
	var width := maxf(SIZE, plate.size.x)
	return Rect2((SIZE - width) * 0.5, 0.0, width, plate.end.y)


## The plate under the name.
func _name_plate() -> Rect2:
	var font := _font()
	var width := font.get_string_size(shown_name(), HORIZONTAL_ALIGNMENT_LEFT, -1, NAME_FONT).x
	var size_with_pad := Vector2(width, font.get_height(NAME_FONT)) + NAME_PAD * 2.0
	return Rect2(Vector2((SIZE - size_with_pad.x) * 0.5, _name_top()), size_with_pad)


## Redraws the frame, icon, badge and chips from the tree.
func refresh() -> void:
	var current := state()
	var hidden := current == &"hidden"
	var lit := current in [&"bought", &"maxed"]
	_icon.visible = not hidden
	_icon.material = null if lit or hidden else grey_material()
	_question.visible = hidden
	badge.visible = not hidden
	_pill.visible = not hidden
	badge.text = "MAX" if current == &"maxed" else "%d/%d" % [tree.level(data.id), data.max_level]
	badge.add_theme_color_override(&"font_color", state_color(current) if lit else UiStyle.TEXT_COLOR)
	_lock.visible = locked()
	free_badge.visible = tree.is_free(data.id)
	ready_chip.visible = can_upgrade()
	crest.visible = current == &"maxed"
	_pill.queue_redraw()
	queue_redraw()


## Picks this node: emits chosen.
func click() -> void:
	chosen.emit(data.id)


## The hover card: the name and what the node gives now.
func tooltip_content() -> Control:
	if state() == &"hidden":
		return TooltipCard.note("???")
	return TooltipCard.note("%s\n%s" % [data.display_name, data.value_text(tree.level(data.id))])


func _gui_input(event: InputEvent) -> void:
	var press := event as InputEventMouseButton
	if press and press.pressed and press.button_index == MOUSE_BUTTON_LEFT:
		Sfx.play(&"click", UiStyle.CLICK_DB)
		click()
		accept_event()


func _set_hovered(on: bool) -> void:
	_hovered = on
	queue_redraw()


func _font() -> Font:
	return get_theme_default_font()


func _name_top() -> float:
	return SIZE - BADGE_TUCK + BADGE_SIZE.y + NAME_GAP


func _draw() -> void:
	var current := state()
	var frame := StyleBoxFlat.new()
	frame.bg_color = HIDDEN_FILL if current == &"hidden" else FILL_COLOR
	frame.set_border_width_all(border_width(current))
	frame.set_corner_radius_all(CORNER)
	frame.border_color = state_color(current)
	if current in [&"bought", &"maxed"]:
		frame.shadow_color = Color(state_color(current), 0.4)
		frame.shadow_size = 16
	else:
		frame.shadow_color = Color(0, 0, 0, 0.55)
		frame.shadow_size = 8
		frame.shadow_offset = Vector2(0, 4)
	draw_style_box(frame, Rect2(0, 0, SIZE, SIZE))
	if selected or _hovered:
		var outline := StyleBoxFlat.new()
		outline.draw_center = false
		outline.border_color = Color.WHITE if selected else Color(1, 1, 1, 0.45)
		outline.set_border_width_all(3 if selected else 2)
		outline.set_corner_radius_all(CORNER + 4)
		draw_style_box(outline, Rect2(-6, -6, SIZE + 12, SIZE + 12))
	var plate := _name_plate()
	var back := StyleBoxFlat.new()
	back.bg_color = NAME_PLATE

## A tapered strip along `points`, `from_width` wide at its start and
## `to_width` at its end: lit on its upper side, shaded under, outlined, and
## with `rim` (moonlight) along its upper edge when given.
static func draw_strip(canvas: CanvasItem, points: PackedVector2Array, from_width: float, to_width: float, color: Color, rim := Color(0, 0, 0, 0)) -> void:
	var count := points.size()
	if count < 2:
		return
	# Which side faces up, for a strip going left or right.
	var flip := -1.0 if points[-1].x < points[0].x else 1.0
	var upper := PackedVector2Array()
	var lower := PackedVector2Array()
	var vertices := PackedVector2Array()
	var colors := PackedColorArray()
	var indices := PackedInt32Array()
	for i in count:
		var along := (points[mini(i + 1, count - 1)] - points[maxi(i - 1, 0)]).normalized()
		var across := along.orthogonal() * flip * lerpf(from_width, to_width, float(i) / (count - 1)) * 0.5
		upper.append(points[i] + across)
		lower.append(points[i] - across)
		vertices.append_array([points[i] + across, points[i] - across])
		colors.append_array([color.lightened(0.14), color.darkened(0.4)])
		if i > 0:
			var a := i * 2 - 2
			indices.append_array([a, a + 1, a + 2, a + 1, a + 3, a + 2])
	RenderingServer.canvas_item_add_triangle_array(canvas.get_canvas_item(), indices, vertices, colors)
	canvas.draw_polyline(upper, BARK_EDGE, 1.5, true)
	canvas.draw_polyline(lower, BARK_EDGE, 1.5, true)
	if rim.a > 0.0:
		canvas.draw_polyline(upper, rim, 1.5, true)


## A leaf from `at`, pointing at `angle`.
static func draw_leaf(canvas: CanvasItem, at: Vector2, angle: float, length: float, color: Color) -> void:
	var along := Vector2.from_angle(angle)
	var across := along.orthogonal()
	var points := PackedVector2Array()
	for k in 5:
		var t := k / 4.0
		points.append(at + along * length * t + across * sin(PI * t) * length * 0.32)
	for k in range(3, 0, -1):
		var t := k / 4.0
		points.append(at + along * length * t - across * sin(PI * t) * length * 0.32)
	canvas.draw_colored_polygon(points, color)


## `points` resampled every `step` px along the line.
static func resample(points: PackedVector2Array, step: float) -> PackedVector2Array:
	var out := PackedVector2Array()
	if points.is_empty():
		return out
	out.append(points[0])
	var carry := 0.0
	for i in range(1, points.size()):
		var from := points[i - 1]
		var length := from.distance_to(points[i])
		var at := step - carry
		while at <= length:
			out.append(from.lerp(points[i], at / length))
			at += step
		carry = length - (at - step)
	if out[-1] != points[-1]:
		out.append(points[-1])
	return out


## Whether a circle at `center` with `radius` overlaps `rect`.
static func circle_hits_rect(center: Vector2, radius: float, rect: Rect2) -> bool:
	var nearest := Vector2(clampf(center.x, rect.position.x, rect.end.x), clampf(center.y, rect.position.y, rect.end.y))
	return nearest.distance_to(center) < radius

## A wavy line up the bark, `across` of the trunk's width from its middle
## (-0.5 left edge, 0.5 right edge), from `from` up to `to`.
func _bark_line(across: float, from: float, to: float, wave: float) -> PackedVector2Array:
	var line := PackedVector2Array()
	var y := from
	while y > to:
		line.append(Vector2(middle + across * width_at(y) + sin(y * 0.045 + wave) * 2.5, y))
		y -= 10.0
	return line

## A short twig off a bough, up and outward, with a leaf sprig at its tip;
## none where the tip would be near a node.
func _add_twig(points: PackedVector2Array, rng: RandomNumberGenerator) -> void:
	var count := points.size()
	var i := int((count - 1) * rng.randf_range(0.35, 0.7))
	var along := (points[mini(i + 1, count - 1)] - points[maxi(i - 1, 0)]).normalized()
	var turn := rng.randf_range(0.6, 1.0)
	# Up and outward: above a bough going right or going left.
	var direction := along.rotated(-turn if along.x >= 0.0 else turn)
	var tip := points[i] + direction * rng.randf_range(26.0, 40.0)
	if page.near_node(tip):
		return
	twigs.append(PackedVector2Array([points[i], points[i].lerp(tip, 0.5) + direction.orthogonal() * 2.0, tip]))
	sprigs.append(Vector3(tip.x, tip.y, direction.angle()))

## Adds a clump unless it would cover the header or Reset tree or leave the
## page; one reaching the ground shrinks, and one left too small is dropped.
func _add_clump(at: Vector2, radius: float, layer: int, rng: RandomNumberGenerator) -> void:
	radius = minf(radius, TreePage.GROUND_Y - CLUMP_GROUND - at.y)
	if radius < CLUMP_MIN:
		return
	for rect: Rect2 in TreePage.KEEP_CLEAR:
		if circle_hits_rect(at, radius * PUFF_REACH, rect):
			return
	if at.x - radius < 12.0 or at.x + radius > TreePage.SIZE.x - 12.0:
		return
	clumps.append(Vector4(at.x, at.y, radius, layer))
	var bumps := PackedVector3Array()
	for k in rng.randi_range(5, 7):
		var angle := lerpf(-PI * 0.95, -PI * 0.05, (k + rng.randf_range(-0.3, 0.3)) / 6.0)
		bumps.append(Vector3(cos(angle) * radius * 0.78, sin(angle) * radius * 0.78, radius * rng.randf_range(0.32, 0.45)))
	puffs.append(bumps)
	var marks := PackedVector4Array()
	if layer > 0:
		for k in 3:
			var spot := at + Vector2(rng.randf_range(-0.6, 0.6), rng.randf_range(-0.6, 0.4)) * radius
			marks.append(Vector4(spot.x, spot.y, rng.randf() * TAU, rng.randf_range(9.0, 14.0)))
	leaf_marks.append(marks)

## Whether `rect` would touch a node (its frame, badge or name), a Backpack
## preview plaque or the ground.

func _layout() -> void:
	var nodes := tree.branch_nodes(branch)
	var columns := Vector2(INF, -INF)
	var top_row := 0.0
	for data in nodes:
		columns = Vector2(minf(columns.x, data.grid.x), maxf(columns.y, data.grid.x))
		top_row = maxf(top_row, data.grid.y)
	var middle := Vector2((columns.x + columns.y) * 0.5, top_row * 0.5)
	for data in nodes:
		centers[data.id] = Vector2(SIZE.x * 0.5 + (data.grid.x - middle.x) * CELL.x, MIDDLE_Y - (data.grid.y - middle.y) * CELL.y)
	for limb: Dictionary in LIMBS.get(branch, []):
		var ids: Array = limb["ids"]
		for i in ids.size():
			parents[ids[i]] = ids[i - 1] if i > 0 else &""
	for data in nodes:
		if not parents.has(data.id):
			parents[data.id] = &""
	trunk_top = GROUND_Y - LIMB_GROUND
	for id: StringName in parents:
		var parent: StringName = parents[id]
		on_trunk[id] = absf(centers[id].x - SIZE.x * 0.5) < 1.0 and (parent == &"" or on_trunk[parent])
		if on_trunk[id]:
			trunk_top = minf(trunk_top, centers[id].y)
	for id: StringName in parents:
		if not on_trunk[id] and parents[id] == &"":
			trunk_top = minf(trunk_top, _attach_y(id))
	for id: StringName in parents:
		if on_trunk[id]:
			continue
		var parent: StringName = parents[id]
		var from: Vector2
		if parent != &"":
			from = centers[parent]
		else:
			# From the trunk's own side, so boughs going left and right never cross.
			var y := _attach_y(id)
			from = Vector2(SIZE.x * 0.5 + signf(centers[id].x - SIZE.x * 0.5) * trunk_width(y) * 0.3, y)
		limbs[id] = _limb_curve(from, centers[id])


## Where node `id`'s limb leaves the trunk: LIMB_DROP under the node, or
## from behind a node on the trunk that would sit there (its name included).
func _attach_y(id: StringName) -> float:
	var y := minf(centers[id].y + LIMB_DROP, GROUND_Y - LIMB_GROUND)
	for other: StringName in on_trunk:
		var center: Vector2 = centers[other]
		if on_trunk[other] and y > center.y - TreeNodeView.SIZE * 0.5 and y < center.y + TreeNodeView.SIZE:
			return center.y + TreeNodeView.SIZE * 0.2
	return y


## The trunk's width at height `y` (TreeArt adds the root flare).
func trunk_width(y: float) -> float:
	return lerpf(TRUNK_ROOT, TRUNK_TIP, inverse_lerp(GROUND_Y, trunk_top, y))


## A limb from `from` out to `to`, arching like a bough: it rises from the
## trunk, bends outward and comes into the node from the inner side; a node
## straight above gets a straight limb.
func _limb_curve(from: Vector2, to: Vector2) -> PackedVector2Array:
	var d := to - from
	var c1 := from + Vector2(d.x * 0.2, d.y * 0.6 - absf(d.x) * 0.22)
	var c2 := to - Vector2(d.x * 0.45, absf(d.x) * 0.12)
	return bezier(from, c1, c2, to)


static func bezier(p0: Vector2, p1: Vector2, p2: Vector2, p3: Vector2) -> PackedVector2Array:
	var points := PackedVector2Array()
	for i in CURVE_STEPS + 1:
		var t := float(i) / CURVE_STEPS
		var u := 1.0 - t
		points.append(p0 * u * u * u + p1 * 3.0 * u * u * t + p2 * 3.0 * u * t * t + p3 * t * t * t)
	return points


## Points from node `id` down its limbs to the trunk (the last point is on the
## trunk), or only down to `stop` when that is on the way.
func stem(id: StringName, stop := &"") -> PackedVector2Array:
	var points := PackedVector2Array([centers[id]])
	var at := id
	while at != stop and not on_trunk[at]:
		var limb: PackedVector2Array = limbs[at]
		for i in range(limb.size() - 2, -1, -1):
			points.append(limb[i])
		at = parents[at]
		if at == &"":
			return points
	if at != stop and stop != &"":
		# Both on the trunk.
		points.append(centers[stop])
	return points


## Inside a node's frame, badge or name.
func near_node(at: Vector2) -> bool:
	for id: StringName in centers:
		var center: Vector2 = centers[id]
		if Rect2(center - Vector2(CELL.x * 0.5 - 4.0, 60.0), Vector2(CELL.x - 8.0, 156.0)).has_point(at):
			return true
	return false


func _grows_from(id: StringName, ancestor: StringName) -> bool:
	var at: StringName = parents[id]
	while at != &"":
		if at == ancestor:
			return true
		at = parents[at]
	return false


## The vine from node `id` to `required`, along the limbs and the trunk.
func _vine_path(id: StringName, required: StringName) -> PackedVector2Array:
	if _grows_from(id, required) or (on_trunk[id] and on_trunk[required]):
		return stem(id, required)
	var path := stem(id)
	var back := stem(required)
	back.reverse()
	path.append_array(back)
	return path

## A vine winding along `path`, with small leaves; gold when `lit`.
func _draw_vine(path: PackedVector2Array, lit: bool, phase: float) -> void:
	var points := TreeArt.resample(path, 5.0)
	var count := points.size()
	if count < 2:
		return
	var wavy := PackedVector2Array()
	var normals := PackedVector2Array()
	for i in count:
		var along := (points[mini(i + 1, count - 1)] - points[maxi(i - 1, 0)]).normalized()
		normals.append(along.orthogonal())
		wavy.append(points[i] + normals[i] * sin(i * 5.0 / VINE_WAVE * TAU + phase) * VINE_AMP)
	var color := VINE_LIT if lit else VINE_DIM
	if lit:
		draw_polyline(wavy, Color(LIGHT, 0.16), VINE_WIDTH + 7.0, true)
	draw_polyline(wavy, color, VINE_WIDTH, true)
	var every := int(VINE_LEAF_STEP / 5.0)
	for i in range(every, count - every, every):
		var side := 1.0 if (i / every) % 2 == 0 else -1.0
		var along := normals[i].orthogonal() * -1.0
		TreeArt.draw_leaf(self, wavy[i], (along + normals[i] * side * 1.2).angle(), 10.0, color.darkened(0.1))

```glsl
shader_type canvas_item;
// Behind a menu: the screen behind it, blurred, darkened and vignetted.

uniform sampler2D screen_texture : hint_screen_texture, filter_linear;
// How far apart the blur's taps are, in pixels.
uniform float spread = 4.0;
// How much darker the blurred screen gets (0 none, 1 black).
uniform float darken = 0.5;
// How much darker its corners get on top of that.
uniform float vignette = 0.55;

void fragment() {
	vec3 sum = vec3(0.0);
	float total = 0.0;
	// 13 taps in a diamond.
	for (int x = -2; x <= 2; x++) {
		for (int y = -2; y <= 2; y++) {
			if (abs(x) + abs(y) > 2) {
				continue;
			}
			float weight = 1.0 / (1.0 + float(x * x + y * y));
			sum += texture(screen_texture, SCREEN_UV + vec2(float(x), float(y)) * SCREEN_PIXEL_SIZE * spread).rgb * weight;
			total += weight;
		}
	}
	vec3 color = sum / total * (1.0 - darken);
	float edge = smoothstep(0.3, 0.9, length((UV - vec2(0.5)) * vec2(1.3, 1.0)));
	COLOR = vec4(color * (1.0 - vignette * edge), 1.0);
}

static func rounded_rect(rect: Rect2, radius: float, steps := 6) -> PackedVector2Array:
	var points := PackedVector2Array()

	close_popup()
	var opened := SkinPurchasePopup.new(Skins.find(id), SaveGame.instance().essences)
	purchase = opened
	opened.confirmed.connect(_buy)
	opened.closed.connect(func() -> void:
		if purchase == opened:
			purchase = null)
	add_child(opened)


## A hidden shop closes its pop-up, so it never shows on another page.
func _on_visibility_changed() -> void:
	if not visible:
		close_popup()


func _buy(id: StringName) -> void:
	if SaveGame.instance().buy_skin(id):
		Sfx.play(&"coin")


func _equip(id: StringName) -> void:
	SaveGame.instance().equip_skin(id)
```
